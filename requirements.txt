# Plain English Translator — dependencies
#
# Translating .txt documents needs NOTHING beyond the Python standard library.
# Everything below is only required to read other file formats, and is imported
# lazily, so a missing package degrades that one format instead of breaking the
# whole tool.

# PDF support (translator.py tries PyMuPDF first, falls back to PyPDF2)
PyMuPDF>=1.20.0
PyPDF2>=2.0.0

# Word .docx support
python-docx>=0.8.11
